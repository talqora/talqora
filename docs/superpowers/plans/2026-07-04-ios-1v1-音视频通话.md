# iOS 1:1 音视频通话 (Phase 1) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** iOS 原生端实现 1:1 语音+视频通话,与 web 端互通,复用现有 `call:*` Socket.io 信令、coturn TURN、`/api/turn-credentials`。

**Architecture:** 三件套——`WebRTCSession`(actor 封装 `stasel/WebRTC` 的 `RTCPeerConnection`,吐 Sendable 事件流)、`CallFeature`(TCA 状态机,对齐 web `useCall`)、`CallView`(语音/视频共享状态机)。`SocketClient` 扩展 `call:*` 收发。纯逻辑(DTO/解析器/reducer)走 TDD;媒体/UI 走 build + 真机↔web 互通验收。

**Tech Stack:** Swift 6(严格并发)· iOS 18 · Tuist+SPM · TCA · SwiftUI · Swift Testing · `stasel/WebRTC`(libwebrtc)· SocketIO(现有)。

**Spec:** `docs/superpowers/specs/2026-07-04-ios-1v1-音视频通话-design.md`

**分支:** `ios`(iOS 工作按仓库约定走 `ios` 分支)。

**前置约束(全程遵守):**
- 每个任务"代码+测试同提交";纯函数/reducer 高覆盖单测。
- 动任何 UI 前先加载 `mobile-ios-design` + `swiftui-pro` skill;守 mobile-swift/CLAUDE.md UX 硬规范(44pt、`PressableButtonStyle`、`WeChatTheme`/`WeChatFont` 令牌、三态、明暗)。
- 新增文件后必须 `cd mobile-swift && tuist generate`(sources glob 在 generate 时固化,不重生成新文件进不了工程)。
- 完工门禁:`cd mobile-swift && tuist generate && xcodebuild build`(+ 测试 target)。

**wire 字段名(iOS 收发必须逐字匹配,camelCase):**
- `call:start` → `{callId, from:CallUser, to:CallUser, offer:{type,sdp}, callType:"voice"|"video"}`
- `call:accept` → `{callId, from:<int id>, to:<int id>, answer:{type,sdp}}`
- `call:reject` / `call:end` → `{callId}`
- `call:ice` → `{callId, candidate:{candidate:String, sdpMlineIndex?:Int, sdpMid?:String}}`(注意 `sdpMlineIndex` 小写 L)
- `call:rejoin` → `{callId, from:CallUser, to:CallUser, offer:{type,sdp}}`
- `call:busy`/`call:handled`/`call:peer-reconnecting` → 入站,`{callId, ...}`
- `CallUser` → `{id:Int, username:String, nickname:String, avatar:String}`
- `callId` 规则:`call_{fromId}_{toId}_{timestamp}`

```swift
import Foundation

// 把 socket.io 投递的 call:* 原始字典解析成领域类型。纯函数,可单测。
// 复用 SocketMessageParser.intValue 做多形态取整(NSNumber/Int/String)。
enum SocketCallParsers {
    static func parseUser(_ any: Any?) -> CallUserDTO? {
        guard let d = any as? [String: Any], let id = SocketMessageParser.intValue(d["id"]) else { return nil }
        return CallUserDTO(id: id,
                           username: d["username"] as? String ?? "",
                           nickname: d["nickname"] as? String ?? "",
                           avatar: d["avatar"] as? String ?? "")
    }

    static func parseSDP(_ any: Any?) -> SessionDescriptionDTO? {
        guard let d = any as? [String: Any], let type = d["type"] as? String, let sdp = d["sdp"] as? String
        else { return nil }
        return SessionDescriptionDTO(type: type, sdp: sdp)
    }

    static func parseCandidate(_ any: Any?) -> IceCandidateDTO? {
        guard let d = any as? [String: Any], let c = d["candidate"] as? String else { return nil }
        return IceCandidateDTO(candidate: c,
                               sdpMlineIndex: SocketMessageParser.intValue(d["sdpMlineIndex"]),
                               sdpMid: d["sdpMid"] as? String)
    }

    static func parseIncoming(_ raw: Any) -> CallIncoming? {
        guard let d = raw as? [String: Any],
              let callId = d["callId"] as? String,
              let from = parseUser(d["from"]), let to = parseUser(d["to"]),
              let offer = parseSDP(d["offer"]) else { return nil }
        let type = CallType(rawValue: d["callType"] as? String ?? "voice") ?? .voice
        return CallIncoming(callId: callId, from: from, to: to, offer: offer, callType: type)
    }

    static func parseAccept(_ raw: Any) -> (callId: String, answer: SessionDescriptionDTO)? {
        guard let d = raw as? [String: Any], let callId = d["callId"] as? String,
              let answer = parseSDP(d["answer"]) else { return nil }
        return (callId, answer)
    }

    static func parseIce(_ raw: Any) -> (callId: String, candidate: IceCandidateDTO)? {
        guard let d = raw as? [String: Any], let callId = d["callId"] as? String,
              let cand = parseCandidate(d["candidate"]) else { return nil }
        return (callId, cand)
    }

    static func parseRejoin(_ raw: Any) -> CallRejoin? {
        guard let d = raw as? [String: Any], let callId = d["callId"] as? String,
              let from = parseUser(d["from"]), let to = parseUser(d["to"]),
              let offer = parseSDP(d["offer"]) else { return nil }
        return CallRejoin(callId: callId, from: from, to: to, offer: offer)
    }

    static func callId(_ raw: Any) -> String? { (raw as? [String: Any])?["callId"] as? String }
    static func handledStatus(_ raw: Any) -> String? { (raw as? [String: Any])?["status"] as? String }
}
```

```swift
import Dependencies
import DependenciesMacros
import Foundation

struct TurnCredentialsResponse: Codable, Equatable, Sendable {
    var iceServers: [IceServerDTO]
    var ttl: Int
}

@DependencyClient
struct TurnCredentialsClient: Sendable {
    var fetch: @Sendable () async throws -> [IceServerDTO]
}

extension TurnCredentialsClient: DependencyKey {
    static let liveValue = TurnCredentialsClient(fetch: {
        @Dependency(\.keychain) var keychain
        let token = (try? keychain.load(.accessToken)) ?? nil
        var req = URLRequest(url: URL(string: APIEnvironment.current.baseURLString + "/api/turn-credentials")!)
        if let token { req.setValue("Bearer \(token)", forHTTPHeaderField: "Authorization") }
        let (data, _) = try await URLSession.shared.data(for: req)
        return try JSONDecoder().decode(TurnCredentialsResponse.self, from: data).iceServers
    })
    static let previewValue = TurnCredentialsClient(fetch: { [] })
}

extension DependencyValues {
    var turnCredentials: TurnCredentialsClient {
        get { self[TurnCredentialsClient.self] }
        set { self[TurnCredentialsClient.self] = newValue }
    }
}
```
> 鉴权头形式(Bearer vs cookie)以 server `/api/turn-credentials` 的 `authenticateToken` 实现为准;若走 cookie 则改设 `Cookie: token=...`(对齐 SocketClient 做法)。实现时核对 `server/src/routes` 该端点。

```swift
import ComposableArchitecture
import Foundation

@Reducer
struct CallFeature {
    @ObservableState
    struct State: Equatable {
        var phase: CallPhase = .idle
        var callId: String = ""
        var callType: CallType = .voice
        var peer: CallUserDTO?
        var role: CallRole = .caller
        var pendingOffer: SessionDescriptionDTO?
        var isMuted = false
        var isSpeakerOn = false
        var isCameraOn = true
        var isFrontCamera = true
        var hasRemoteVideo = false
        var durationSeconds = 0
        var localUser: CallUserDTO?
    }

    enum Action: Equatable {
        // UI
        case startCall(peer: CallUserDTO, type: CallType)
        case acceptTapped, rejectTapped, hangupTapped
        case toggleMute, toggleSpeaker, toggleCamera, switchCamera
        // 信令入站
        case incomingCall(CallIncoming)
        case remoteAccepted(answer: SessionDescriptionDTO)
        case remoteRejected, remoteEnded, remoteBusy(callId: String)
        case remoteIce(IceCandidateDTO)
        case remotePeerReconnecting
        case remoteRejoin(CallRejoin)
        // 媒体事件
        case localCandidate(IceCandidateDTO)
        case remoteVideoAvailable
        case connectionState(String)
        // 内部
        case tick
        case failed(String)
    }

    @Dependency(\.webRTCSession) var webRTC
    @Dependency(\.socketClient) var socket
    @Dependency(\.turnCredentials) var turn
    @Dependency(\.continuousClock) var clock

    private enum CancelID { case events, timer }

    var body: some ReducerOf<Self> {
        Reduce { state, action in
            switch action {
            case let .incomingCall(ev):
                state.phase = .incoming; state.callId = ev.callId; state.peer = ev.from
                state.callType = ev.callType; state.role = .callee; state.pendingOffer = ev.offer
                return .none

            case .rejectTapped:
                let id = state.callId
                state.phase = .ended(reason: "已拒绝")
                return .merge(.run { _ in socket.sendCallReject(id) }, cleanup())

            case let .remoteBusy(callId) where callId == state.callId:
                state.phase = .ended(reason: "对方忙线中")
                return cleanup()

            case .hangupTapped, .remoteEnded:
                let id = state.callId
                state.phase = .ended(reason: "通话结束")
                return .merge(.run { _ in socket.sendCallEnd(id) }, cleanup())

            case .remoteRejected:
                state.phase = .ended(reason: "对方已拒绝")
                return cleanup()

            // …… 其余迁移(startCall/acceptTapped/remoteAccepted/ICE/media/tick/toggle*)
            //    按 §7 数据流实现:见下方“完整迁移”注记。
            default:
                return .none
            }
        }
    }

    private func cleanup() -> Effect<Action> {
        .merge(.cancel(id: CancelID.events), .cancel(id: CancelID.timer),
               .run { _ in await webRTC.close() })
    }
}
```

> **完整迁移(实现时按此补全 `default` 之外的分支,每补一支加一条 TestStore 测试再实现):**
> - `startCall`:置 `.outgoing`+生成 callId → effect:`turn.fetch → webRTC.configure → startLocalMedia(video) → createOffer → socket.sendCallStart`;并起 `events` 订阅(见下)与计时。
> - `acceptTapped`:若 `pendingOffer` 存在 → `.connecting` → effect:`turn.fetch → configure → startLocalMedia → createAnswer(pendingOffer) → socket.sendCallAccept`;否则(刷新后无 offer)走 rejoin(发 `call:rejoin` 新 offer)。
> - `remoteAccepted`:`webRTC.setRemoteAnswer(answer)`,`.connecting`。
> - `remoteIce`:`webRTC.addRemoteCandidate`。
> - `localCandidate`:`socket.sendCallIce`。
> - `connectionState("connected")`:`.connected` + 起 `tick` 计时;`"failed"` → `.failed`。
> - `remoteVideoAvailable`:`hasRemoteVideo=true`。
> - `toggleMute/Speaker/Camera/switchCamera`:改 state + 调 `webRTC.setMuted/...`。
> - `tick`:`durationSeconds += 1`。
> - `remotePeerReconnecting`:`.reconnecting`;`remoteRejoin`:按被叫重协商(`createAnswer` 回 `call:accept`)。
> - 订阅 effect(`.run`,id `.events`):`for await ev in webRTC.events()` → send 对应 media Action;`for await ev in socket.events()` 过滤 `call*` → send 对应信令 Action(**两条流各一个 `.run`,合并**)。

```swift
import WebRTC

private actor RTCEngine {
    private static let factory: RTCPeerConnectionFactory = {
        RTCInitializeSSL()
        return RTCPeerConnectionFactory(encoderFactory: RTCDefaultVideoEncoderFactory(),
                                        decoderFactory: RTCDefaultVideoDecoderFactory())
    }()
    private var pc: RTCPeerConnection?
    private var audioTrack: RTCAudioTrack?
    private var pendingCandidates: [IceCandidateDTO] = []   // remoteDescription 未设前暂存(对齐 web §9)
    private var continuation: AsyncStream<WebRTCEvent>.Continuation?
    private var relayOnly = false
    private var iceServers: [IceServerDTO] = []
    private let delegate = PCDelegate()

    func setContinuation(_ c: AsyncStream<WebRTCEvent>.Continuation) { continuation = c; delegate.sink = { [weak self] e in Task { await self?.ingest(e) } } }

    func configure(_ servers: [IceServerDTO], relayOnly: Bool) {
        self.iceServers = servers; self.relayOnly = relayOnly; buildPC()
    }
    private func buildPC() {
        let cfg = RTCConfiguration()
        cfg.iceServers = iceServers.map { RTCIceServer(urlStrings: $0.urls, username: $0.username, credential: $0.credential) }
        cfg.sdpSemantics = .unifiedPlan
        if relayOnly { cfg.iceTransportPolicy = .relay }
        let constraints = RTCMediaConstraints(mandatoryConstraints: nil, optionalConstraints: nil)
        pc = Self.factory.peerConnection(with: cfg, constraints: constraints, delegate: delegate)
    }

    func startLocalMedia(video: Bool) {
        let source = Self.factory.audioSource(with: RTCMediaConstraints(mandatoryConstraints: nil, optionalConstraints: nil))
        let track = Self.factory.audioTrack(with: source, trackId: "audio0")
        audioTrack = track
        pc?.add(track, streamIds: ["stream0"])
        // 视频轨在 Task 12 补(RTCCameraVideoCapturer)。
    }

    func createOffer() async throws -> SessionDescriptionDTO {
        let c = RTCMediaConstraints(mandatoryConstraints: ["OfferToReceiveAudio": "true", "OfferToReceiveVideo": "true"], optionalConstraints: nil)
        let sdp = try await pc!.offer(for: c)
        try await pc!.setLocalDescription(sdp)
        return SessionDescriptionDTO(type: "offer", sdp: sdp.sdp)
    }
    func createAnswer(remoteOffer: SessionDescriptionDTO) async throws -> SessionDescriptionDTO {
        try await pc!.setRemoteDescription(RTCSessionDescription(type: .offer, sdp: remoteOffer.sdp))
        drainPending()
        let c = RTCMediaConstraints(mandatoryConstraints: ["OfferToReceiveAudio": "true", "OfferToReceiveVideo": "true"], optionalConstraints: nil)
        let sdp = try await pc!.answer(for: c)
        try await pc!.setLocalDescription(sdp)
        return SessionDescriptionDTO(type: "answer", sdp: sdp.sdp)
    }
    func setRemoteAnswer(_ answer: SessionDescriptionDTO) async throws {
        try await pc!.setRemoteDescription(RTCSessionDescription(type: .answer, sdp: answer.sdp))
        drainPending()
    }
    func addRemoteCandidate(_ dto: IceCandidateDTO) {
        guard let pc, pc.remoteDescription != nil else { pendingCandidates.append(dto); return }
        pc.add(RTCIceCandidate(sdp: dto.candidate, sdpMLineIndex: Int32(dto.sdpMlineIndex ?? 0), sdpMid: dto.sdpMid))
    }
    private func drainPending() {
        let items = pendingCandidates; pendingCandidates = []
        for dto in items { addRemoteCandidate(dto) }
    }
    func reset() { let keep = pendingCandidates; pc?.close(); pc = nil; buildPC(); pendingCandidates = keep }
    func setMuted(_ m: Bool) { audioTrack?.isEnabled = !m }
    func close() { pc?.close(); pc = nil; continuation?.finish() }

    private func ingest(_ e: WebRTCEvent) { continuation?.yield(e) }
}

// delegate 在 WebRTC 信令线程回调;立即转 Sendable DTO 再经 sink 送回 actor。
private final class PCDelegate: NSObject, RTCPeerConnectionDelegate, @unchecked Sendable {
    var sink: (@Sendable (WebRTCEvent) -> Void)?
    func peerConnection(_ pc: RTCPeerConnection, didGenerate c: RTCIceCandidate) {
        sink?(.localCandidate(IceCandidateDTO(candidate: c.sdp, sdpMlineIndex: Int(c.sdpMLineIndex), sdpMid: c.sdpMid)))
    }
    func peerConnection(_ pc: RTCPeerConnection, didChange s: RTCPeerConnectionState) {
        let name: String = { switch s { case .connected: "connected"; case .failed: "failed"; case .closed: "closed"; default: "other" } }()
        sink?(.connectionState(name))
    }
    func peerConnection(_ pc: RTCPeerConnection, didAdd r: RTCRtpReceiver, streams: [RTCMediaStream]) {
        if r.track?.kind == "video" { sink?(.remoteVideoAvailable) }
    }
    // 其余必需空实现:didChange signalingState/iceGatheringState/iceConnectionState/didRemove/didOpen dataChannel/shouldNegotiate
}
```
